## 🔍 Code Snippet

```go
package main

import (
    "log"
    "os"

    engine "github.com/1mcatch/lcnc-security-engine"
    "github.com/1mcatch/lcnc-security-engine/model"
)

func main() {
    workflowJSON, err := os.ReadFile("workflow.json")
    if err != nil {
        log.Fatal(err)
    }

    report, err := engine.New().Analyze(model.PlatformN8N, workflowJSON)
    if err != nil {
        log.Fatal(err)
    }

    log.Printf("risk score: %.0f", report.Score)
}
```
