```cs
using Microsoft.VisualStudio.TestPlatform.Utilities;
using System.ComponentModel;
using System.Linq.Expressions;
using System.Reflection;
using Throw;
using Xunit.Sdk;

namespace dinq.tests;

public static class Dinq
{
    public static IQueryable<T> Execute<T>(this IQueryable<T> q, string query)
    {
        var type = typeof(T);
        ParameterExpression p = Expression.Parameter(type, "p");

        if (string.IsNullOrWhiteSpace(query))
            return q.Where(Noop<T>(p));

        var s = query.Split();
        var (prop, op, value) = (s[0], s[1], s[2]);

        var pred = Predicate<T>(p, prop, op, value);
        var combinedPredicate = new Expression[] { pred }.Aggregate(Expression.AndAlso);
```
