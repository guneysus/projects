using Newtonsoft.Json;
using StackExchange.Redis;
using System;
using System.Collections;
using System.Collections.Generic;

namespace redis_backed_dictionary
{
    public class RedisBackedDict<V> : IDictionary<string, V>
    {
        private readonly ConnectionMultiplexer _redis;
        private readonly IDatabase _db;

        public RedisBackedDict()
        {
            this._redis = ConnectionMultiplexer.Connect("localhost");
            this._db = _redis.GetDatabase(db: -1);

        }

        public V this[string key] { get {
                return JsonConvert.DeserializeObject<V>(_db.StringGet(key));
            } set => throw new NotImplementedException(); }

        public ICollection<string> Keys => throw new NotImplementedException();
